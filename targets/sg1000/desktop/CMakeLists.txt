cmake_minimum_required(VERSION 3.20)
project(sg1000_desktop LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
find_package(SDL2 CONFIG REQUIRED)
option(SG_STATIC_SDL "Link SDL2 statically" OFF)
set(SG_EMBEDDED_ROM_HEADER "" CACHE FILEPATH "Generated embedded cartridge header")
set(SG ${CMAKE_CURRENT_SOURCE_DIR}/..)
add_executable(sg1000_play ${SG}/host/sg_player.c ${SG}/runtime/z80.c
    ${SG}/runtime/sg1000.c ${SG}/runtime/sg_audio.c ${SG}/runtime/tms9918.c)
target_include_directories(sg1000_play PRIVATE ${SG}/runtime)
if(TARGET SDL2::SDL2main)
    target_link_libraries(sg1000_play PRIVATE SDL2::SDL2main)
endif()
if(SG_STATIC_SDL)
    target_link_libraries(sg1000_play PRIVATE SDL2::SDL2-static)
else()
    target_link_libraries(sg1000_play PRIVATE SDL2::SDL2)
endif()
if(SG_EMBEDDED_ROM_HEADER)
    get_filename_component(ROM_INCLUDE_DIR "${SG_EMBEDDED_ROM_HEADER}" DIRECTORY)
    target_include_directories(sg1000_play PRIVATE "${ROM_INCLUDE_DIR}")
    target_compile_definitions(sg1000_play PRIVATE SG_EMBEDDED_ROM)
endif()
if(NOT MSVC)
    target_compile_options(sg1000_play PRIVATE -Wall -Wextra)
endif()
# The archive ships its own SDL2 beside the executable on Linux.
set_target_properties(sg1000_play PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
install(TARGETS sg1000_play RUNTIME DESTINATION .)
